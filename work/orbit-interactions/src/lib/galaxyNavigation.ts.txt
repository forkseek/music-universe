export const ALBUM_TARGET = '@album-star'
export const MIN_SCENE_ZOOM = 0.82
export const MAX_SCENE_ZOOM = 2.8
export const OVERVIEW_ENTER = 1.72
export const OVERVIEW_EXIT = 1.42

export interface GalaxyView {
  mode: 'overview' | 'focus'
  target: string
  zoom: number
  transitioning: boolean
  gestureUntil: number
}

export function initialGalaxyView(): GalaxyView {
  return { mode: 'overview', target: ALBUM_TARGET, zoom: OVERVIEW_ENTER + 0.08, transitioning: false, gestureUntil: 0 }
}

/** Separate view state from seeded data and the scene animation clock. */
export function zoomGalaxyView(view: GalaxyView, pixels: number): GalaxyView {
  if (!Number.isFinite(pixels) || pixels === 0) return view
  const zoom = Math.max(MIN_SCENE_ZOOM, Math.min(MAX_SCENE_ZOOM, view.zoom * Math.exp(Math.max(-240, Math.min(240, pixels)) * 0.00165)))
  const overview = view.mode === 'overview' ? zoom > OVERVIEW_EXIT : zoom >= OVERVIEW_ENTER
  return { ...view, zoom, mode: overview ? 'overview' : 'focus' }
}

// Each event has a fresh R3F raycast target; a previous hover never chooses a planet.
const wheelTargets = new WeakMap<Event, string>()
export function markWheelTarget(event: Event, target: string) { wheelTargets.set(event, target) }
export function readWheelTarget(event: WheelEvent) {
  const hit = wheelTargets.get(event)
  if (hit !== undefined) return hit
  return event.target instanceof Element ? event.target.closest<HTMLElement>('[data-scene-target]')?.dataset.sceneTarget : undefined
}
